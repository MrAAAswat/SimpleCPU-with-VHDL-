LIBRARY IEEE;
USE IEEE.STD_LOGIC_1164.ALL;
USE IEEE.NUMERIC_STD.ALL;
USE work.MyPackage.ALL; -- Assumes HexDisplay component is here

ENTITY SampleCPU IS
    PORT (
        -- Clock
        clk    : IN STD_LOGIC;
        -- Switches Input
        Opcode : IN STD_LOGIC_VECTOR(3 DOWNTO 0);
        Rs     : IN STD_LOGIC_VECTOR(1 DOWNTO 0);
        Rt     : IN STD_LOGIC_VECTOR(1 DOWNTO 0);
        Data   : IN STD_LOGIC_VECTOR(7 DOWNTO 0);
        -- Hex Output
        HEX0, HEX1, HEX2, HEX3, HEX4, HEX5 : OUT STD_LOGIC_VECTOR(6 DOWNTO 0);
        -- Debug LEDs
        LEDG : OUT STD_LOGIC_VECTOR(3 DOWNTO 0)
    );
END SampleCPU;

ARCHITECTURE Behavioral OF SampleCPU IS

    -- Register File Definitions
    TYPE reg_array IS ARRAY (0 TO 3) OF STD_LOGIC_VECTOR(7 DOWNTO 0);
    SIGNAL reg_file : reg_array := (OTHERS => (OTHERS => '0')); -- r0 ~ r3
    
    -- Pipeline Valid Bits (Tracking active instructions)
    SIGNAL valid : STD_LOGIC_VECTOR(3 DOWNTO 0) := (OTHERS => '0');

    -- IF/ID Pipeline Registers
    SIGNAL IF_ID_Data   : STD_LOGIC_VECTOR(7 DOWNTO 0);
    SIGNAL IF_ID_Opcode : STD_LOGIC_VECTOR(3 DOWNTO 0);
    SIGNAL IF_ID_Rs     : STD_LOGIC_VECTOR(1 DOWNTO 0);
    SIGNAL IF_ID_Rt     : STD_LOGIC_VECTOR(1 DOWNTO 0);

    -- ID/EXE Pipeline Registers
    SIGNAL ID_EXE_Opcode  : STD_LOGIC_VECTOR(3 DOWNTO 0);
    SIGNAL ID_EXE_Rs      : STD_LOGIC_VECTOR(1 DOWNTO 0);
    SIGNAL ID_EXE_Rt      : STD_LOGIC_VECTOR(1 DOWNTO 0);
    SIGNAL ID_EXE_Rs_Data : STD_LOGIC_VECTOR(7 DOWNTO 0);
    SIGNAL ID_EXE_Rt_Data : STD_LOGIC_VECTOR(7 DOWNTO 0);

    -- EXE/WB Pipeline Registers
    SIGNAL EXE_WB_Rs     : STD_LOGIC_VECTOR(1 DOWNTO 0);
    SIGNAL EXE_WB_Result : STD_LOGIC_VECTOR(7 DOWNTO 0);

    -- Debug Signals
    SIGNAL rs_seg, rt_seg : STD_LOGIC_VECTOR(7 DOWNTO 0);

BEGIN

    -- ==========================================
    -- Stage 1: Instruction Fetch (IF)
    -- ==========================================
    PROCESS (clk)
    BEGIN
        IF rising_edge(clk) THEN
            IF Opcode = "1111" THEN -- NOP Check
                valid(0) <= '0';
            ELSE
                valid(0) <= '1';
                IF_ID_Data   <= Data;
                IF_ID_Opcode <= Opcode;
                IF_ID_Rs     <= Rs;
                IF_ID_Rt     <= Rt;
            END IF;
        END IF;
    END PROCESS;

    -- ==========================================
    -- Stage 2: Instruction Decode (ID) & Reg Read
    -- ==========================================
    PROCESS (clk)
    BEGIN
        IF rising_edge(clk) THEN
            IF valid(0) = '0' THEN
                valid(1) <= '0';
            ELSE
                valid(1) <= '1';
                -- Pass control signals
                ID_EXE_Opcode <= IF_ID_Opcode;
                ID_EXE_Rs     <= IF_ID_Rs;
                ID_EXE_Rt     <= IF_ID_Rt;

                -- Read Operands
                IF IF_ID_Opcode = "0000" THEN -- Load Immediate (Data -> Rs)
                    ID_EXE_Rs_Data <= IF_ID_Data;
                ELSE
                    ID_EXE_Rs_Data <= reg_file(to_integer(unsigned(IF_ID_Rs)));
                END IF;
                
                ID_EXE_Rt_Data <= reg_file(to_integer(unsigned(IF_ID_Rt)));
            END IF;
        END IF;
    END PROCESS;

    -- ==========================================
    -- Stage 3: Execution (EXE)
    -- ==========================================
    PROCESS (clk)
        VARIABLE result : STD_LOGIC_VECTOR(7 DOWNTO 0) := (OTHERS => '0');
    BEGIN
        IF rising_edge(clk) THEN
            IF valid(1) = '0' THEN
                valid(2) <= '0';
            ELSE
                valid(2) <= '1';
                
                CASE ID_EXE_Opcode IS
                    WHEN "0000" => -- Load (Pass immediate through)
                        result := ID_EXE_Rs_Data;
                        
                    WHEN "0001" => -- Move (Rs <= Rt)
                        result := ID_EXE_Rt_Data;
                        
                    WHEN "0010" => -- Add
                        result := STD_LOGIC_VECTOR(unsigned(ID_EXE_Rs_Data) + unsigned(ID_EXE_Rt_Data));
                        
                    WHEN "0011" => -- And
                        result := ID_EXE_Rs_Data AND ID_EXE_Rt_Data;
                        
                    WHEN "0101" => -- Sub: Rs - Rt
                        result := STD_LOGIC_VECTOR(unsigned(ID_EXE_Rs_Data) - unsigned(ID_EXE_Rt_Data));
                        
                    WHEN "1001" => -- Sub: Rt - Rs
                        result := STD_LOGIC_VECTOR(unsigned(ID_EXE_Rt_Data) - unsigned(ID_EXE_Rs_Data));
                        
                    WHEN "0110" => -- Nor
                        result := NOT (ID_EXE_Rs_Data OR ID_EXE_Rt_Data);
                        
                    WHEN "0100" => -- Slt (Set Less Than)
                        IF unsigned(ID_EXE_Rs_Data) < unsigned(ID_EXE_Rt_Data) THEN
                            result := "00000001";
                        ELSE
                            result := "00000000";
                        END IF;
                        
                    WHEN "1000" => -- Div
                        -- Division is complex for single cycle. 
                        -- Suggest using a specific LPM_DIV or IP core here.
                        result := (OTHERS => '0'); 
                        
                    WHEN OTHERS =>
                        result := (OTHERS => '0');
                END CASE;

                EXE_WB_Rs     <= ID_EXE_Rs;
                EXE_WB_Result <= result;
            END IF;
        END IF;
    END PROCESS;

    -- ==========================================
    -- Stage 4: Write Back (WB)
    -- ==========================================
    PROCESS (clk)
    BEGIN
        IF rising_edge(clk) THEN
            IF valid(2) = '0' THEN
                valid(3) <= '0';
            ELSE
                valid(3) <= '1';
                reg_file(to_integer(unsigned(EXE_WB_Rs))) <= EXE_WB_Result;
            END IF;
        END IF;
    END PROCESS;

    -- ==========================================
    -- Outputs
    -- ==========================================
    LEDG <= valid; -- Show which stages are active

    -- Asynchronous read for display purposes (shows current state of registers)
    rs_seg <= reg_file(to_integer(unsigned(Rs)));
    rt_seg <= reg_file(to_integer(unsigned(Rt)));

    -- Hex Display Instantiations
    BusHex0 : HexDisplay PORT MAP(Hex => Data(3 DOWNTO 0), Seg => HEX0);
    BusHex1 : HexDisplay PORT MAP(Hex => Data(7 DOWNTO 4), Seg => HEX1);
    RsHex0  : HexDisplay PORT MAP(Hex => rs_seg(3 DOWNTO 0), Seg => HEX2);
    RsHex1  : HexDisplay PORT MAP(Hex => rs_seg(7 DOWNTO 4), Seg => HEX3);
    RtHex0  : HexDisplay PORT MAP(Hex => rt_seg(3 DOWNTO 0), Seg => HEX4);
    RtHex1  : HexDisplay PORT MAP(Hex => rt_seg(7 DOWNTO 4), Seg => HEX5);

END Behavioral;
